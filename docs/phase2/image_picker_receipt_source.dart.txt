import 'dart:io';

import 'package:image_picker/image_picker.dart';
import 'package:permission_handler/permission_handler.dart';

import 'receipt_image_source.dart';

class ImagePickerReceiptSource implements ReceiptImageSource {
  ImagePickerReceiptSource({ImagePicker? picker})
    : _picker = picker ?? ImagePicker();

  final ImagePicker _picker;

  @override
  Future<ReceiptImagePick> pickFromCamera() async {
    final granted = await Permission.camera.request();
    if (!granted.isGranted) {
      return const ReceiptImagePick.failure('permission_denied');
    }

    final image = await _picker.pickImage(
      source: ImageSource.camera,
      imageQuality: 85,
      maxWidth: 1920,
    );
    return image == null
        ? const ReceiptImagePick.failure('cancelled')
        : ReceiptImagePick.path(image.path);
  }

  @override
  Future<ReceiptImagePick> pickFromGallery() async {
    if (Platform.isIOS) {
      final status = await Permission.photos.request();
      if (!status.isGranted && !status.isLimited) {
        return const ReceiptImagePick.failure('permission_denied');
      }
    }

    final image = await _picker.pickImage(
      source: ImageSource.gallery,
      imageQuality: 85,
      maxWidth: 1920,
    );
    return image == null
        ? const ReceiptImagePick.failure('cancelled')
        : ReceiptImagePick.path(image.path);
  }
}
